# Core runtime dependencies.
#
# Pinned to versions verified working on Python 3.13. The previous pins
# (numpy 1.26.2, scikit-learn 1.3.2, torch 2.1.2) predate 3.13 wheels and
# forced a source build that fails without a C toolchain. Line 28 also carried
# an inline "--index-url", which pip rejects on a requirement line, so the file
# could not be installed at all.
#
# Optional extras (training, explainability, streaming, dashboard) live in
# setup.py; install them with e.g. `pip install -e ".[ml]"`.

# API framework. Versions chosen to clear the advisories pip-audit reports:
# starlette < 1.3.1 and python-multipart < 0.0.31 both carry open CVEs.
fastapi==0.141.1
starlette==1.3.1
uvicorn[standard]==0.34.0
pydantic==2.11.3
pydantic-settings==2.3.0
python-multipart==0.0.32

# Database. aiosqlite backs the default SQLite URL; greenlet is required by
# SQLAlchemy's async engine. Add asyncpg for PostgreSQL.
sqlalchemy==2.0.46
aiosqlite==0.19.0
greenlet==3.3.0
alembic==1.13.1

# Cache
redis==5.2.1

# Authentication. PyJWT rather than python-jose: python-jose 3.3.0 has
# unpatched advisories and pulls in ecdsa (itself unpatched) and rsa for
# asymmetric algorithms this service does not use. Only HS256 is needed.
PyJWT==2.13.0

# Numerics. Required by the fingerprinting and analytics helpers; joblib loads
# trained model artefacts when MODEL_PATH contains any.
numpy==1.26.4
pandas==2.2.3
joblib==1.5.2

# Observability
prometheus-client==0.19.0
structlog==23.2.0
